Bài 19. Robot di chuyển hình vuông bằng encoder

Bài học liên thông sử dụng cấu hình robot mới nhất và thư viện hàm dùng chung.

1. Mục tiêu bài học

  • Đi thẳng một quãng đường theo số xung thay vì thời gian.
  • Quay phải 90° theo encoder.
  • Phát hiện kẹt bánh hoặc mất encoder bằng timeout.
Điều kiện học trước: Bài 6, 13, 14 và 15.

2. Chuẩn bị

Phần cứng: Hai động cơ, hai encoder D2/D12 và D3/D14, OLED.

Phần mềm: Arduino IDE, board Arduino Mega or Mega 2560, Serial Monitor 115200 baud.

Trước khi biên dịch, chép thư mục THU_VIEN_ROBOT/RobotMega2560Mini vào Documents/Arduino/libraries, sau đó khởi động lại Arduino IDE. Kê hai bánh khỏi mặt đất ở lần thử đầu tiên.

3. Nguyên lý và thuật toán

  1. Lặp bốn lần: đi một cạnh, dừng giảm trượt, quay phải 90°.
  2. Mỗi chuyển động kết thúc khi hai encoder đạt số xung đặt trước.
  3. Nếu hết thời gian mà chưa đủ xung, robot dừng an toàn và báo lỗi.

4. Chương trình hoàn chỉnh

Chương trình sử dụng thư viện chung RobotMega2560Mini. Tải file .ino
Bai_19_Hinh_vuong_encoder.ino
#include <Robot.h>

// BÀI 19: ROBOT DI CHUYỂN HÌNH VUÔNG BẰNG ENCODER
// Robot đi bốn cạnh bằng số xung encoder, sau mỗi cạnh quay phải 90 độ.
// Hai hằng SO_XUNG_MOT_CANH và SO_XUNG_QUAY_90 phải đo lại trên robot thật.

const long SO_XUNG_MOT_CANH = 300;
const long SO_XUNG_QUAY_90 = 82;
const int PWM_DI = 105;
const int PWM_QUAY = 105;
const unsigned long TIMEOUT_DI_MS = 7000UL;
const unsigned long TIMEOUT_QUAY_MS = 3500UL;

bool daChayXong = false;
bool coLoi = false;

void hienThiHinhVuong(const char *trangThai, uint8_t canh) {
  long trai, phai;
  docHaiEncoder(trai, phai);

  oledXoa();
  oledTieuDe("BAI 19 - HINH VUONG");
  oledInChu(4, 17, trangThai);
  oledInChu(4, 31, "CANH:"); oledInSo(39, 31, canh);
  oledInChu(55, 31, "/4");
  oledInChu(4, 46, "E T/P:");
  oledInSo(45, 46, trai); oledInSo(86, 46, phai);
  oledCapNhat();

  Serial.print(F("Canh=")); Serial.print(canh); Serial.print(F("/4  "));
  Serial.print(trangThai); Serial.print(F("  Encoder T/P="));
  Serial.print(trai); Serial.print('/'); Serial.println(phai);
}

void baoLoi(const char *noiDung, uint8_t canh) {
  dungRobot();
  coLoi = true;
  hienThiHinhVuong(noiDung, canh);
  Serial.println(F("Robot dung an toan. Kiem tra encoder hoac vat can."));
}

void setup() {
  batDauSerial();
  oledBatDau();
  dongCoBatDau();
  encoderBatDau();
  oledThongBao("BAI 19", "HINH VUONG ENCODER",
               "DAT ROBOT LEN SAN", "CHO 2 GIAY");
  delay(2000);
}

void loop() {
  if (daChayXong || coLoi) {
    dungRobot();
    return;
  }

  for (uint8_t canh = 1; canh <= 4; canh++) {
    hienThiHinhVuong("DI MOT CANH", canh);
    if (!diTheoSoXung(SO_XUNG_MOT_CANH, PWM_DI, TIMEOUT_DI_MS)) {
      baoLoi("LOI DI THANG", canh);
      return;
    }
    hienThiHinhVuong("DUNG GIAM TRUOT", canh);
    delay(350);

    hienThiHinhVuong("QUAY PHAI 90", canh);
    if (!quayPhaiTheoSoXung(SO_XUNG_QUAY_90, PWM_QUAY,
                            TIMEOUT_QUAY_MS)) {
      baoLoi("LOI QUAY 90", canh);
      return;
    }
    delay(350);
  }

  dungRobot();
  daChayXong = true;
  hienThiHinhVuong("HOAN THANH", 4);
  Serial.println(F("Da di xong hinh vuong. Nhan RESET de chay lai."));
}

5. Phân tích chương trình

  • Số xung của một cạnh phụ thuộc đường kính bánh và độ phân giải encoder.
  • Số xung quay 90° phụ thuộc khoảng cách hai bánh và ma sát mặt sàn.
  • Dừng 350 ms sau mỗi pha giúp bánh giảm quán tính trước lệnh tiếp theo.

6. Quy trình thực hành và hiệu chỉnh

  1. Đo số xung khi robot đi đúng 30 cm để đặt SO_XUNG_MOT_CANH.
  2. Đánh dấu hướng ban đầu, chỉnh SO_XUNG_QUAY_90 cho đúng góc.
  3. Chạy đủ bốn cạnh và đo sai lệch vị trí cuối.
  4. Thử giữ một bánh để xác nhận timeout dừng robot.

7. Bài tập mở rộng

  1. Viết hàm diHinhChuNhat(dai, rong).
  2. Cho robot quay trái thay vì quay phải.
  3. Tính số xung tự động từ kích thước bánh và khoảng cách hai bánh.

8. Lỗi thường gặp và cách xử lý

Góc tăng dần sau mỗi cạnh: giảm tốc hoặc thêm hiệu chỉnh sai số tích lũy.
Một encoder đếm ngược: kiểm tra hướng encoder trong RobotConfig/RobotEncoder.
Robot chạy mãi: phải giữ timeout và kê bánh khi thử lần đầu.

9. Kết luận
