Bài 9. Hai cảm biến đo xa laser VL53L0X

Khởi tạo hai VL53L0X cùng loại bằng XSHUT, đổi địa chỉ I2C và đo khoảng cách vuông góc về hai bên trái–phải.

1. Mục tiêu bài học

  • Giải thích được nguyên nhân hai VL53L0X trùng địa chỉ mặc định 0x29.
  • Bật lần lượt cảm biến bằng XSHUT và đặt địa chỉ 0x30, 0x31.
  • Đọc khoảng cách trái–phải theo đúng vị trí lắp thực tế.
  • Phát hiện lỗi khởi tạo và phép đo ngoài tầm.

2. Kiến thức trọng tâm

Hai VL53L0X dùng chung bus I2C D20/D21 và cùng có địa chỉ mặc định 0x29. Phải giữ cả hai ở trạng thái tắt, bật cảm biến trái để đổi thành 0x30, sau đó mới bật cảm biến phải và đổi thành 0x31.

Cảm biến trái có XSHUT D39 và nhìn sang trái; cảm biến phải có XSHUT D37 và nhìn sang phải. Ánh xạ này đã được dùng trong cấu hình thư viện mới nhất.

Nếu đổi thứ tự XSHUT hoặc nhầm trái–phải, thuật toán mê cung sẽ quay về phía gần vật cản hơn.

4. Chương trình thực hành

Mã ví dụ sử dụng đúng cấu hình đã kiểm tra trên robot, có OLED và Serial để quan sát.

Bài 9 – đo độc lập hai laser trái/phải
// BÀI 9: HAI CẢM BIẾN LASER VL53L0X
#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>
#include <Adafruit_VL53L0X.h>

const uint8_t XSHUT_TRAI = 39;
const uint8_t XSHUT_PHAI = 37;
const uint8_t DIA_CHI_TRAI = 0x30;
const uint8_t DIA_CHI_PHAI = 0x31;

Adafruit_SSD1306 oled(128, 64, &Wire, -1);
Adafruit_VL53L0X laserTrai;
Adafruit_VL53L0X laserPhai;

void dungBaoLoi(const __FlashStringHelper *loi) {
  Serial.println(loi);
  oled.clearDisplay();
  oled.setTextColor(SSD1306_WHITE);
  oled.setTextSize(1);
  oled.setCursor(0, 10);
  oled.println(F("LOI KHOI TAO:"));
  oled.println(loi);
  oled.display();
  while (true) delay(1000);
}

void khoiTaoHaiLaser() {
  pinMode(XSHUT_TRAI, OUTPUT);
  pinMode(XSHUT_PHAI, OUTPUT);
  digitalWrite(XSHUT_TRAI, LOW);
  digitalWrite(XSHUT_PHAI, LOW);
  delay(10);

  digitalWrite(XSHUT_TRAI, HIGH);
  delay(10);
  if (!laserTrai.begin(DIA_CHI_TRAI)) {
    dungBaoLoi(F("LASER TRAI D39"));
  }

  digitalWrite(XSHUT_PHAI, HIGH);
  delay(10);
  if (!laserPhai.begin(DIA_CHI_PHAI)) {
    dungBaoLoi(F("LASER PHAI D37"));
  }
}

bool docLaser(Adafruit_VL53L0X &laser, uint16_t &mm) {
  VL53L0X_RangingMeasurementData_t ketQua;
  laser.rangingTest(&ketQua, false);
  if (ketQua.RangeStatus != 0) {
    mm = 0;
    return false;
  }
  mm = ketQua.RangeMilliMeter;
  return true;
}

void hienThi(uint16_t trai, bool okTrai,
             uint16_t phai, bool okPhai) {
  oled.clearDisplay();
  oled.drawRect(0, 0, 128, 64, SSD1306_WHITE);
  oled.setTextColor(SSD1306_WHITE);
  oled.setTextSize(1);
  oled.setCursor(8, 4); oled.println(F("LASER TRAI / PHAI"));
  oled.drawLine(0, 16, 127, 16, SSD1306_WHITE);
  oled.setTextSize(2);
  oled.setCursor(4, 23); oled.print(F("L:"));
  okTrai ? oled.print(trai) : oled.print(F("--"));
  oled.setCursor(4, 44); oled.print(F("R:"));
  okPhai ? oled.print(phai) : oled.print(F("--"));
  oled.setTextSize(1);
  oled.setCursor(103, 50); oled.print(F("mm"));
  oled.display();
}

void setup() {
  Serial.begin(115200);
  Wire.begin();
  if (!oled.begin(SSD1306_SWITCHCAPVCC, 0x3C)) {
    Serial.println(F("LOI OLED 0x3C"));
    while (true) delay(1000);
  }
  khoiTaoHaiLaser();
}

void loop() {
  uint16_t trai, phai;
  bool okTrai = docLaser(laserTrai, trai);
  bool okPhai = docLaser(laserPhai, phai);

  Serial.print(F("Trai="));
  okTrai ? Serial.print(trai) : Serial.print(F("LOI"));
  Serial.print(F(" mm  Phai="));
  okPhai ? Serial.print(phai) : Serial.print(F("LOI"));
  Serial.println(F(" mm"));

  hienThi(trai, okTrai, phai, okPhai);
  delay(100);
}

5. Phân tích chương trình

  • Hai cảm biến được tắt đồng thời trước khi bắt đầu để tránh trùng địa chỉ 0x29.
  • Laser trái D39 được đổi thành 0x30 trước; laser phải D37 được đổi thành 0x31 sau.
  • RangeStatus khác 0 được giữ là lỗi phép đo và hiển thị --.
  • Hàm đọc dùng tham chiếu nên có thể tái sử dụng cho cả hai đối tượng cảm biến.

6. Quy trình thực hành

  1. Cài Adafruit VL53L0X và Adafruit BusIO trong Library Manager.
  2. Đặt vật bên trái rồi bên phải, xác nhận nhãn L/R không bị đảo.
  3. Thử tháo từng dây XSHUT để quan sát thông báo lỗi đúng cảm biến.
  4. Đo tại 100, 200, 400 mm và ghi sai số.

7. Bài tập mở rộng

  1. Tính chênh lệch khoảng cách trái–phải.
  2. Xác định bên có khoảng trống lớn hơn.
  3. Lọc trung vị ba lần đo của mỗi laser.

8. Lỗi thường gặp và cách xử lý

Cả hai laser không khởi tạo: kiểm tra SDA D20, SCL D21 và nguồn.
Chỉ cảm biến thứ hai lỗi: kiểm tra quy trình đổi địa chỉ và XSHUT D37.
Trái/phải bị ngược: xác nhận D39 là trái, D37 là phải trên robot thật.

9. Kết luận

Robot đã có hai khoảng cách bên đúng theo cấu hình thực tế, sẵn sàng cho bài bám tường và mê cung.
