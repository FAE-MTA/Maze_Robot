Bài 13. Encoder và đo tốc độ

Đếm xung bằng ngắt ngoài, xác định chiều quay và tính tốc độ vòng/phút của hai bánh xe.

1. Mục tiêu bài học

  • Giải thích được hai kênh A/B của encoder và cách xác định chiều quay.
  • Đếm xung bằng ngắt ngoài trên D2 và D3 mà không bỏ xung khi vòng lặp bận.
  • Chụp dữ liệu biến dùng chung với ISR một cách an toàn.
  • Tính RPM từ số xung, thời gian lấy mẫu và số xung trên một vòng.

2. Kiến thức trọng tâm

Encoder tạo chuỗi xung khi trục quay. Kênh A được nối tới chân ngắt ngoài; tại mỗi cạnh lên của A, mức logic kênh B cho biết hướng quay tương đối. Biến đếm phải khai báo volatile vì được thay đổi trong hàm ngắt.

Công thức: RPM = Δxung × 60.000 / (xung mỗi vòng × Δt ms). Hằng số xung mỗi vòng phải được kiểm tra theo cách đếm một kênh hay hai kênh và vị trí encoder trước/sau hộp số.

Hàm ngắt phải thật ngắn: chỉ đọc kênh B và tăng/giảm biến đếm. Mọi phép tính, OLED và Serial đều thực hiện ở loop().

4. Chương trình thực hành

Chương trình dưới đây được viết đầy đủ, có chú thích tiếng Việt và có thể mở trực tiếp trong Arduino IDE.

Bài 13 – quay bánh bằng tay và đo RPM
// BÀI 13: ĐỌC ENCODER VÀ TÍNH RPM
#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>

const uint8_t ENC_TRAI_A = 2;
const uint8_t ENC_TRAI_B = 12;
const uint8_t ENC_PHAI_A = 3;
const uint8_t ENC_PHAI_B = 14;

// Hiệu chỉnh theo encoder và phương pháp đếm thực tế.
const float XUNG_MOI_VONG = 20.0;
const int8_t HE_SO_CHIEU_TRAI = 1;
const int8_t HE_SO_CHIEU_PHAI = -1;

volatile long xungTrai = 0;
volatile long xungPhai = 0;

Adafruit_SSD1306 oled(128, 64, &Wire, -1);

void ngatEncoderTrai() {
  int huong = digitalRead(ENC_TRAI_B) ? 1 : -1;
  xungTrai += huong * HE_SO_CHIEU_TRAI;
}

void ngatEncoderPhai() {
  int huong = digitalRead(ENC_PHAI_B) ? 1 : -1;
  xungPhai += huong * HE_SO_CHIEU_PHAI;
}

void chupSoXung(long &trai, long &phai) {
  noInterrupts();
  trai = xungTrai;
  phai = xungPhai;
  interrupts();
}

void setup() {
  Serial.begin(115200);
  pinMode(ENC_TRAI_A, INPUT_PULLUP);
  pinMode(ENC_TRAI_B, INPUT_PULLUP);
  pinMode(ENC_PHAI_A, INPUT_PULLUP);
  pinMode(ENC_PHAI_B, INPUT_PULLUP);

  attachInterrupt(digitalPinToInterrupt(ENC_TRAI_A),
                  ngatEncoderTrai, RISING);
  attachInterrupt(digitalPinToInterrupt(ENC_PHAI_A),
                  ngatEncoderPhai, RISING);

  Wire.begin();
  if (!oled.begin(SSD1306_SWITCHCAPVCC, 0x3C)) {
    Serial.println("LOI OLED");
    while (true) delay(1000);
  }
  Serial.println("QUAY TUNG BANH BANG TAY");
}

void loop() {
  static unsigned long mocMs = millis();
  static long xungTraiCu = 0;
  static long xungPhaiCu = 0;

  if (millis() - mocMs >= 500) {
    unsigned long bayGio = millis();
    unsigned long dt = bayGio - mocMs;
    mocMs = bayGio;

    long trai, phai;
    chupSoXung(trai, phai);
    long deltaTrai = trai - xungTraiCu;
    long deltaPhai = phai - xungPhaiCu;
    xungTraiCu = trai;
    xungPhaiCu = phai;

    float rpmTrai = deltaTrai * 60000.0 / (XUNG_MOI_VONG * dt);
    float rpmPhai = deltaPhai * 60000.0 / (XUNG_MOI_VONG * dt);

    Serial.print("XUNG T/P: ");
    Serial.print(trai); Serial.print(" / "); Serial.print(phai);
    Serial.print("  RPM T/P: ");
    Serial.print(rpmTrai, 1); Serial.print(" / "); Serial.println(rpmPhai, 1);

    oled.clearDisplay();
    oled.setTextColor(SSD1306_WHITE);
    oled.setTextSize(1);
    oled.setCursor(0, 0);  oled.println("ENCODER QUAY TAY");
    oled.setCursor(0, 18); oled.print("Xung T: "); oled.println(trai);
    oled.setCursor(0, 30); oled.print("Xung P: "); oled.println(phai);
    oled.setCursor(0, 44); oled.print("RPM T: "); oled.println(rpmTrai, 1);
    oled.setCursor(0, 56); oled.print("RPM P: "); oled.println(rpmPhai, 1);
    oled.display();
  }
}

5. Phân tích chương trình và thuật toán

  • Ngắt chỉ xử lý cạnh lên kênh A nên số xung/vòng phải tương ứng đúng chế độ đếm này.
  • Hai biến đếm được chụp trong vùng tạm khóa ngắt, tránh đọc dở một biến long đang được ISR cập nhật.
  • RPM có dấu thể hiện chiều quay. Nếu quay bánh theo chiều tiến mà RPM âm, đổi đúng hệ số chiều của bánh đó.
  • Chu kỳ thực tế dt được dùng trong công thức, do đó sai số do thời gian hiển thị OLED được giảm.

6. Quy trình thực hành

  1. Không cấp PWM cho động cơ; kê robot và quay bánh trái bằng tay đúng chiều tiến.
  2. Kiểm tra xung trái tăng và RPM trái dương; lặp lại với bánh phải.
  3. Đánh dấu bánh xe, quay đúng một vòng và kiểm tra số xung để hiệu chỉnh XUNG_MOI_VONG.
  4. Quay hai bánh ở tốc độ khác nhau và đối chiếu RPM trên OLED.

7. Bài tập củng cố và mở rộng

  1. Đếm cả hai cạnh CHANGE của kênh A và xác định lại số xung/vòng.
  2. Phát hiện encoder hỏng khi động cơ có PWM nhưng không xuất hiện xung trong 500 ms.
  3. Vẽ đồ thị RPM theo thời gian bằng Serial Plotter.

8. Lỗi thường gặp và cách xử lý

Xung luôn bằng 0: kiểm tra nguồn encoder, D2/D3 và lựa chọn cạnh ngắt.
Xung tăng khi quay cả hai chiều: kiểm tra dây kênh B D12/D14 hoặc logic xác định chiều.
RPM sai theo một hệ số cố định: XUNG_MOI_VONG chưa đúng với hộp số và chế độ đếm.
Không gọi Serial, OLED hoặc delay() bên trong hàm ngắt.

9. Kết luận

Bài học tạo được phép đo tốc độ độc lập cho hai bánh. Bài 14 dùng số xung để đo quãng đường, còn Bài 15 dùng chênh lệch xung để giữ robot đi thẳng.
